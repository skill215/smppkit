func main(){

 "00000010 00000015 00000000 00000001",
 "00000021 00000004 00000000 00000007 0000000000000000000000000000040000",

 "00000024 00000103 00000000 00000007 00000000000000000000 0427000102 0427000102",

for _,v:=range []pdu.Version{pdu.V33,pdu.V34,pdu.V50}{for _,mode:=range []message.Mode{message.UDH8,message.UDH16,message.SAR,message.Payload}{for radio:=1;radio<=140;radio++{for _,n:=range []int{0,1,radio-1,radio,radio+1,2*radio,300}{for _,extra:=range []int{-1,0,1,4,6,120,248}{for _,port:=range []bool{false,true}{
 data:=bytes.Repeat([]byte{255},n);msg:=message.Binary{Data:data,DataCoding:4,Source:pdu.Address{Value:"S"},Destination:pdu.Address{Value:"D"},ServiceType:"svc",Optional:tlv.Fields{{Tag:0x1400,Value:[]byte{1,2,3}}}}
 if v==pdu.V33{msg.Optional=nil};if port{msg.Optional=append(msg.Optional,tlv.Uint16(tlv.SourcePort,42))}

}}}}}}

}
