func reviewLiteral(t *testing.T, s string) []byte {
	t.Helper()
	b, err := hex.DecodeString(s)
	if err != nil {
		t.Fatal(err)
	}
	return b
}

// Independent data_sm bytes, empty mandatory fields and two customer TLVs.
func TestFinalReviewDecodedSliceIsolation(t *testing.T) {
	raw := reviewLiteral(t, "000000240000010300000000000000010000000000000000000015000001111501000122")
	for _, v := range []pdu.Version{pdu.V34, pdu.V50} {
		t.Run(fmt.Sprintf("%x", v), func(t *testing.T) {
			c, err := codec.New(codec.Config{Version: v})
			if err != nil {
				t.Fatal(err)
			}
			packet, err := c.Decode(raw)
			if err != nil {
				t.Fatal(err)
			}
			fields := packet.Body.(*pdu.DataSM).Optional
			before := bytes.Clone(fields[1].Value)
			fields[0].Value[0] = 0x33
			if !bytes.Equal(fields[1].Value, before) || raw[len(raw)-1] != 0x22 {
				t.Fatal("control: in-range edit changed another field/input")
			}
			fields[0].Value = append(fields[0].Value, 1, 2, 3, 4, 5)
			if !bytes.Equal(fields[1].Value, before) {
				t.Fatalf("appending TLV[0] changed TLV[1]: %x -> %x", before, fields[1].Value)
			}
		})
	}
}

// A literal peer submits a transaction request to the automatic memory ACK
// helper. No delivery worker is started and Receiver.Receive is never called.
func TestFinalReviewBufferedTransactionACK(t *testing.T) {
	for _, v := range []pdu.Version{pdu.V34, pdu.V50} {
		for _, command := range []string{"data_sm", "submit_sm"} {
			for _, mode := range []byte{2, 3} {
				if v == pdu.V34 && command == "submit_sm" {
					continue
				} // 3.4 transaction is data_sm-only
				t.Run(fmt.Sprintf("%x/%s/mode%d", v, command, mode), func(t *testing.T) {
					ctx, cancel := context.WithTimeout(t.Context(), time.Second)
					defer cancel()
					c, err := codec.New(codec.Config{Version: v})
					if err != nil {
						t.Fatal(err)
					}
					a, b := net.Pipe()
					s, err := newSession(a, c, Config{}, true, Bound, Transceiver)
					if err != nil {
						t.Fatal(err)
					}
					defer func() { s.Close(); b.Close(); s.Wait() }()
					receiver, err := message.NewReceiver(message.ReceiveConfig{AckPolicy: message.AckOnBuffered})
					if err != nil {
						t.Fatal(err)
					}
					frame := "0000002000000103000000000000000100000000000000020000042400024869"
					if command == "submit_sm" {
						frame = "0000002300000004000000000000000100000000000000020000000000000000024869"
					}
					raw := reviewLiteral(t, frame)
					raw[23] = mode // esm_class after empty service/source/destination fields
					done := make(chan error, 1)
					go func() { _, err := b.Write(raw); done <- err }()
					in, err := s.Receive(ctx)
					if err != nil {
						t.Fatal(err)
					}
					if err := <-done; err != nil {
						t.Fatal(err)
					}
					ack := make(chan pdu.PDU, 1)
					ackErr := make(chan error, 1)
					go func() { p, e := c.Read(b); ack <- p; ackErr <- e }()
					err = AdmitMessage(ctx, receiver, in, "")
					if err != nil {
						if mode == 3 {
							t.Fatal("store-and-forward control rejected:", err)
						}
						return
					} // corrected helper may reject the unsupported mode
					response := <-ack
					if err := <-ackErr; err != nil {
						t.Fatal(err)
					}
					if mode == 3 {
						if response.Status != pdu.OK || receiver.Stats().Queued != 1 {
							t.Fatal("store-and-forward control failed")
						}
						return
					}
					if response.Status == pdu.OK {
						t.Fatalf("transaction acknowledged OK before any delivery/consumption; queue=%d", receiver.Stats().Queued)
					}
				})
			}
		}

	}
}
